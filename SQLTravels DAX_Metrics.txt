01. Executive KPIs
------------------

Total Revenue =
SUM(vrlt_fact_booking[FareAmount])

Total Bookings =
COUNTROWS(vrlt_fact_booking)

Total Passengers =
SUM(vrlt_fact_booking[SeatsBooked])

Total Trips =
COUNTROWS(vrlt_fact_trip)

Average Fare per Passenger =
DIVIDE([Total Revenue], [Total Passengers])

Average Fare per Booking =
DIVIDE([Total Revenue], [Total Bookings])

Total Operating Cost =
[Total Fuel Cost] + [Total Maintenance Cost]

Gross Profit =
[Total Revenue] - [Total Operating Cost]

Gross Profit Margin % =
DIVIDE([Gross Profit], [Total Revenue])

05. Trip Operations
-------------------

Completed Trips =
CALCULATE(
    [Total Trips],
    vrlt_fact_trip[TripStatus] = "Completed"
)

Delayed Trips =
CALCULATE(
    [Total Trips],
    vrlt_fact_trip[TripStatus] = "Delayed"
)

Cancelled Trips =
CALCULATE(
    [Total Trips],
    vrlt_fact_trip[TripStatus] = "Cancelled"
)

Trip Cancellation Rate =
DIVIDE([Cancelled Trips], [Total Trips])

Delayed Trip Rate =
DIVIDE([Delayed Trips], [Total Trips])

Average Delay Minutes =
AVERAGE(vrlt_fact_trip[DelayMinutes])

On-Time Trips =
CALCULATE(
    [Total Trips],
    vrlt_fact_trip[DelayMinutes] <= 15
)

On-Time Performance % =
DIVIDE([On-Time Trips], [Total Trips])

Revenue per Trip =
DIVIDE([Total Revenue], [Total Trips])

Passengers per Trip =
DIVIDE([Total Passengers], [Total Trips])


06. Occupancy Analytics
-----------------------

Total Seat Capacity =
SUMX(
    vrlt_fact_trip,
    RELATED(vrlt_dim_bus[Capacity])
)

Occupancy % =
DIVIDE([Total Passengers], [Total Seat Capacity])

Available Seats =
[Total Seat Capacity] - [Total Passengers]

High Occupancy Trips =
COUNTROWS(
    FILTER(
        VALUES(vrlt_fact_trip[TripID]),
        [Occupancy %] >= 0.80
    )
)

Sold Out Trips =
COUNTROWS(
    FILTER(
        VALUES(vrlt_fact_trip[TripID]),
        [Occupancy %] >= 0.95
    )
)

Low Occupancy Trips =
COUNTROWS(
    FILTER(
        VALUES(vrlt_fact_trip[TripID]),
        [Occupancy %] < 0.50
    )
)

17. Booking Date Analytics
--------------------------

Bookings by Booking Date =
CALCULATE(
    [Total Bookings],
    USERELATIONSHIP(
        vrlt_fact_booking[BookingDateID],
        vrlt_dim_date[DateID]
    )
)

Revenue by Booking Date =
CALCULATE(
    [Total Revenue],
    USERELATIONSHIP(
        vrlt_fact_booking[BookingDateID],
        vrlt_dim_date[DateID]
    )
)

Passengers by Booking Date =
CALCULATE(
    [Total Passengers],
    USERELATIONSHIP(
        vrlt_fact_booking[BookingDateID],
        vrlt_dim_date[DateID]
    )
)

Average Advance Booking Days =
AVERAGEX(
    vrlt_fact_booking,
    VAR BookingDate =
        DATE(
            INT(vrlt_fact_booking[BookingDateID] / 10000),
            INT(MOD(vrlt_fact_booking[BookingDateID], 10000) / 100),
            MOD(vrlt_fact_booking[BookingDateID], 100)
        )
    VAR TravelDate =
        DATE(
            INT(vrlt_fact_booking[TravelDateID] / 10000),
            INT(MOD(vrlt_fact_booking[TravelDateID], 10000) / 100),
            MOD(vrlt_fact_booking[TravelDateID], 100)
        )
    RETURN
        DATEDIFF(BookingDate, TravelDate, DAY)
)

Notes
-----

1. Use vrlt_dim_date[DateID] -> vrlt_fact_booking[TravelDateID] as the active relationship.
2. Keep vrlt_dim_date[DateID] -> vrlt_fact_booking[BookingDateID] inactive.
3. Use USERELATIONSHIP measures for Booking Date analysis.
4. Keep dimensions filtering facts with single-direction relationships.
5. Avoid active fact-to-fact relationships unless strictly needed.
6. Format percentage measures as Percentage.
7. Format amount measures using Indian units: Lakhs and Crores.
